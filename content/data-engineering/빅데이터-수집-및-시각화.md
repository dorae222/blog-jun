            ```python
            def get_link(key_word, page_range): # 매개변수: 키워드, 페이지 범위
                link = []
            
                for page in range(page_range):
                    current_page = 1 + page*10 # 네이버 뉴스 URL: 1 → 11 → 21 → ...
                    crawling_url_list = URL_BEFORE_KEYWORD + key_word + URL_BEFORE_PAGE_NUM + str(current_page)
                    
                    # URL에 get 요청을 보냄( >>200 → 서버에서 잘 처리되어 OK 싸인을 보냄)
                    response = requests.get(crawling_url_list) # URL에 get 요청을 보냄
            
                    # 응답의 내용은 HTML코드, 'lxml'파서를 이용하여 분석
                    soup = BeautifulSoup(response.text,'lxml') # href는 키, 링크는 값을 의미
            
                    # HTML 문서에서 각 URL을 갖고 있는 태그 선택 & 저장
                    url_tag = soup.select('div.news_area > a')
                    # soup.select('상위태그명 > 하위태그명> 하위태그명')
                    # select를 하면 찾은 태그들을 리스트 형태로 반환
            
                    # url_tag에 저장되어 있는 각 가사의 URL을 link리스트에 저장
                    for url in url_tag:
                        link.append(url['href']) # href: 연결할 주소를 지정하는 속성
            
                return link
            ```

        - wordcount — txt 파일에서 명사만 추출해 빈도순으로 정렬한 뒤 txt 파일로 저장하는 함수
            
            ```python
            def wordcount(file1, file2):
                f = open(file1, 'r', encoding='utf8')
                g = open(file2, 'w', encoding='utf8') # 단어와 개수를 저장하는 파일 생성
                
                engine = Okt() # 형태소 분석기
                data = f.read()
                all_nouns = engine.nouns(data) # 명사 추출
                nouns = [n for n in all_nouns if len(n)> 1] # 리스트 컴프리헨션: 1글자는 의미 없으니 삭제
            
                global count, by_num
            
                count = Counter(nouns) # 명사 개수 세기
                by_num = OrderedDict(sorted(count.items(), key = lambda t: t[1], reverse=True)) # count.items의 값을 기준으로 내림차순 정렬
                word = [i for i in by_num.keys()]
                number = [j for j in by_num.values()]
            
                for w,n in zip(word, number):
                    final1 = '%s  %d' % (w,n)
                    g.write(final1 + '\n')
                print('- 단어 카운팅이 완료되었습니다. (wordcount.txt)\n')
            
                f.close(), g.close()
                return
            ```

            def wordcloud(filename): # filename = crawling2.txt
                print('워드 클라우드 생성 중...')
            
                with open(filename, encoding='utf8') as f:
                    data = f.read()
            
                    engine = Okt()
                    all_nouns = engine.nouns(data)
            
                    nouns = [n for n in all_nouns if len(n)>1]
                    count = Counter(nouns)
            
                    tags = count.most_common(100)
            
                    wc = WordCloud(font_path='malgun',background_color=(168,237,244), width=2500,height=1500)
                    cloud = wc.generate_from_frequencies(dict(tags))
            
                    plt.imshow(cloud, interpolation='bilinear') # 2차원 데이터는 imshow 명령어를 사용하여 시각화
                    plt.axis('off') # 축을 보여주지 않음
                    plt.savefig('cloud.jpg')
                    plt.show()
                print('- cloud.jpg가 저장되었습니다.')
